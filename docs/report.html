8 扩展数据图 ED5：有机碳分子组分相关性热图

# 碳组分数据
comps <- data1[, c("Carbohydrate_fraction", "Protein_fraction", "Lignin_fraction",
                    "Lipid_fraction", "Char_fraction", "Carbonyl_fraction")]
names(comps) <- c("Carbohydrate", "Protein", "Lignin", "Lipid", "Char", "Carbonyl")

# 计算相关矩阵
NMR_cor_vars <- as.matrix(cor(comps, use = "pairwise.complete.obs"))

# 获取上三角
get_upper_tri <- function(cormat) {
  cormat[lower.tri(cormat)] <- NA
  return(cormat)
}

# 重排序
reorder_cormat <- function(cormat) {
  dd <- as.dist((1 - cormat) / 2)
  hc <- hclust(dd)
  cormat <- cormat[hc$order, hc$order]
}

cormat <- reorder_cormat(NMR_cor_vars)
cormat_upper <- get_upper_tri(cormat)

# 将矩阵转换为长格式
NMR_cormat_upper <- as.data.frame(cormat_upper) %>%
  mutate(X1 = row.names(cormat_upper)) %>%
  pivot_longer(cols = -X1, names_to = "X2", values_to = "r") %>%
  filter(!is.na(r))

# 设置因子水平
NMR_cormat_upper$X2 <- factor(NMR_cormat_upper$X2,
                               levels = c("Carbonyl", "Lignin", "Char", "Lipid", "Protein", "Carbohydrate"))
NMR_cormat_upper$X1 <- factor(NMR_cormat_upper$X1,
                               levels = c("Carbonyl", "Lignin", "Char", "Lipid", "Protein", "Carbohydrate"))

# 计算相关系数矩阵
r <- cor(comps, use = "pairwise.complete.obs")

# 四舍五入相关系数
NMR_cormat_upper$r <- round(NMR_cormat_upper$r, 2)
NMR_cormat_upper$Corr <- NMR_cormat_upper$r
NMR_cormat_upper$Corr[NMR_cormat_upper$Corr < 0] <- -1
NMR_cormat_upper$Corr[NMR_cormat_upper$Corr > 0] <- 1
NMR_cormat_upper$Corr[NMR_cormat_upper$Corr == 1] <- "black"
NMR_cormat_upper$Corr[NMR_cormat_upper$Corr == -1] <- "white"

# 绘制热图
plot_corr_map <- ggplot(data = NMR_cormat_upper) +
  geom_tile(aes(x = X2, y = X1, fill = r)) +
  scale_fill_viridis() +
  scale_y_discrete("") +
  scale_x_discrete("") +
  theme(axis.text.x = element_text(angle = 45, hjust = 1, size = 14),
        axis.title.x = element_blank(),
        axis.text.y = element_text(size = 14)) +
  geom_text(data = NMR_cormat_upper,
            mapping = aes(x = X2, y = X1, label = r),
            color = NMR_cormat_upper$Corr)

print(plot_corr_map)
