3 图 1：土壤有机碳组分箱线图

使用 ggplot2 绘制土壤有机碳官能团与分子类别的箱线图，展示不同碳组分的相对丰度分布。

# 加载数据
data <- read.csv(here("data/raw/Hall_Ye_etal_NEON_NMR_biogeo.csv"), header = TRUE)

# 准备碳组分数据
test <- c(data$Carbohydrate_fraction, data$Protein_fraction, data$Lignin_fraction,
          data$Lipid_fraction, data$Char_fraction, data$Carbonyl_fraction)

trts <- c(rep("Carbohydrate", nrow(data)), rep("Protein", nrow(data)),
          rep("Lignin", nrow(data)), rep("Lipid", nrow(data)),
          rep("Char", nrow(data)), rep("Carbonyl", nrow(data)))
trts <- as.factor(trts)
SOM <- data.frame(test, trts)

# 绘制箱线图
p1 <- ggplot(SOM, aes(x = trts, y = test, fill = trts)) +
  geom_boxplot(outlier.shape = NA) +
  geom_jitter(width = 0.2, alpha = 0.5) +
  scale_fill_brewer(palette = "Set2") +
  labs(x = "Carbon Group", y = "Relative Abundance") +
  theme_bw(base_size = 14) +
  theme(legend.position = "none",
        axis.text.x = element_text(angle = 45, hjust = 1))

print(p1)

5 扩展数据图 ED2：环境因子分布箱线图

使用 ggplot2 绘制关键环境变量的分布。

# 选择环境因子列（使用实际存在的列名）
env_cols <- c("phH2o", "carbonTot_g_kg", "nitrogenTot_g_kg", "OC_N_ratio",
              "sandTotal_percent", "siltTotal_percent", "clayTotal_percent",
              "MAP_mm", "MAT_C")

# 转换为长格式
env_data <- data[, env_cols]
env_long <- reshape(env_data, direction = "long",
                    varying = names(env_data),
                    v.names = "Value", timevar = "Variable",
                    times = names(env_data))

# 绘制分面箱线图
p2 <- ggplot(env_long, aes(x = Variable, y = Value)) +
  geom_boxplot(fill = "steelblue", alpha = 0.7) +
  facet_wrap(~Variable, scales = "free", ncol = 3) +
  theme_bw(base_size = 12) +
  theme(axis.text.x = element_text(angle = 45, hjust = 1),
        strip.text = element_text(face = "bold"))

print(p2)
